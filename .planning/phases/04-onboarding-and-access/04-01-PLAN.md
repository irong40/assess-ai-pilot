-- Currently assigns new users to default company with 'viewer' role
-- Needs update to: create a NEW company per signup, set trial_ends_at and trial_status
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE default_company_id uuid;
BEGIN
  SELECT id INTO default_company_id FROM companies LIMIT 1;
  IF default_company_id IS NULL THEN
    INSERT INTO companies (name, description) VALUES ('Default Organization', 'Auto-created organization')
    RETURNING id INTO default_company_id;
  END IF;
  INSERT INTO public.profiles (id, email, company_id, first_name, last_name, role) VALUES (
    new.id, new.email, default_company_id, ...
  );
  RETURN new;
END; $$;
