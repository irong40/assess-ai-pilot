       ```sql
       CREATE TABLE public.agent_settings (
         id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
         company_id UUID NOT NULL REFERENCES public.companies(id) ON DELETE CASCADE,
         agent_type TEXT NOT NULL,
         settings JSONB NOT NULL DEFAULT '{}',
         updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
         created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
         UNIQUE(company_id, agent_type)
       );
       ALTER TABLE public.agent_settings ENABLE ROW LEVEL SECURITY;
       CREATE POLICY "Users view own company settings" ON public.agent_settings FOR SELECT TO authenticated
         USING (company_id = (SELECT p.company_id FROM public.profiles p WHERE p.id = auth.uid()));
       CREATE POLICY "Admin/ISSM update settings" ON public.agent_settings FOR ALL TO authenticated
         USING (company_id = (SELECT p.company_id FROM public.profiles p WHERE p.id = auth.uid()))
         WITH CHECK (company_id = (SELECT p.company_id FROM public.profiles p WHERE p.id = auth.uid()));
       ```
